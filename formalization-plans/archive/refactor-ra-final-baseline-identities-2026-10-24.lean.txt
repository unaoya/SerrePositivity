open Lean in
run_cmd do
  let env ← getEnv
  let mut rows : Array Json := #[]
  let mut repoNames : Array Json := #[]
  for (name, _) in env.constants.toList do
    if `SerrePositivity |>.isPrefixOf name then
      repoNames := repoNames.push (toJson name.toString)
    let some idx := env.getModuleIdxFor? name | continue
    let mod := env.header.modules[idx.toNat]!.module
    unless `SerrePositivity.InfinityCategories.F1B |>.isPrefixOf mod do continue
    rows := rows.push <| Json.mkObj [("name", toJson name.toString), ("module", toJson mod.toString)]
  IO.FS.writeFile "/tmp/r-a-final-round/kernel-f1b-inventory.json" (Json.mkObj
    [("declarations", Json.arr rows), ("repository_names", Json.arr repoNames)]).pretty
  logInfo m!"Exported {rows.size} F1B declarations and {repoNames.size} repository names."
